633: local function kind_of(obj)
634:   if type(obj) ~= 'table' then return type(obj) end
635:   local i = 1
636:   for _ in pairs(obj) do
637:     if obj[i] ~= nil then i = i + 1 else return 'table' end
638:   end
639:   if i == 1 then return 'table' else return 'array' end

642: local function escape_str(s)
643:   local in_char  = {'\\', '"', '/', '\b', '\f', '\n', '\r', '\t'}
644:   local out_char = {'\\', '"', '/',  'b',  'f',  'n',  'r',  't'}
645:   for i, c in ipairs(in_char) do
646:     s = s:gsub(c, '\\' .. out_char[i])
647:   end
648:   return s
649: end

651: local function skip_delim(str, pos, delim, err_if_missing)
652:   pos = pos + #str:match('^%s*', pos)
653:   if str:sub(pos, pos) ~= delim then
654:     return pos, false
655:   end
656:   return pos + 1, true
657: end
658: 
659: local function parse_str_val(str, pos, val)
660:   val = val or ''
661:   local early_end_error = 'End of input found while parsing string.'
662:   local c = str:sub(pos, pos)
663:   if c == '"'  then return val, pos + 1 end
664:   if c ~= '\\' then return parse_str_val(str, pos + 1, val .. c) end
665:   -- We must have a \ character.
666:   local esc_map = {b = '\b', f = '\f', n = '\n', r = '\r', t = '\t'}
667:   local nextc = str:sub(pos + 1, pos + 1)
668:   return parse_str_val(str, pos + 2, val .. (esc_map[nextc] or nextc))
669: end
670: 
671: local function parse_num_val(str, pos)
672:   local num_str = str:match('^-?%d+%.?%d*[eE]?[+-]?%d*', pos)
673:   local val = tonumber(num_str)
674:   return val, pos + #num_str
675: end

677: function jstringify(obj, as_key)
678:   local s = {}
679:   local kind = kind_of(obj)
680:   if kind == 'array' then
681:     s[#s + 1] = '['
682:     for i, val in ipairs(obj) do
683:       if i > 1 then s[#s + 1] = ', ' end
684:       s[#s + 1] = jstringify(val)
685:     end
686:     s[#s + 1] = ']'
687:   elseif kind == 'table' then
688:     s[#s + 1] = '{'
689:     for k, v in pairs(obj) do
690:       if #s > 1 then s[#s + 1] = ', ' end
691:       s[#s + 1] = jstringify(k, true)
692:       s[#s + 1] = ':'
693:       s[#s + 1] = jstringify(v)
694:     end
695:     s[#s + 1] = '}'
696:   elseif kind == 'string' then
697:     return '"' .. escape_str(obj) .. '"'
698:   elseif kind == 'number' then
699:     if as_key then return '"' .. tostring(obj) .. '"' end
700:     return tostring(obj)
701:   elseif kind == 'boolean' then
702:     return tostring(obj)
703:   elseif kind == 'nil' then
704:     return 'null'
705:   end
706:   return table.concat(s)
707: end

711: function jparse(str, pos, end_delim)
712:   pos = pos or 1
713:   if not str then error('Array was not found.') end
714:   local pos = pos + #str:match('^%s*', pos)
715:   local first = str:sub(pos, pos)
716:   if first == '{' then  -- Parse an object.
717:     local obj, key, delim_found = {}, true, true
718:     pos = pos + 1
719:     while true do
720:       key, pos = jparse(str, pos, '}')
721:       if key == nil then return obj, pos end
722:       pos = skip_delim(str, pos, ':', true)
723:       obj[key], pos = jparse(str, pos)
724:       pos, delim_found = skip_delim(str, pos, ',')
725:     end
726:   elseif first == '[' then  -- Parse an array.
727:     local arr, val, delim_found = {}, true, true
728:     pos = pos + 1
729:     while true do
730:       val, pos = jparse(str, pos, ']')
731:       if val == nil then return arr, pos end
732:       arr[#arr + 1] = val
733:       pos, delim_found = skip_delim(str, pos, ',')
734:     end
735:   elseif first == '"' then  -- Parse a string.
736:     return parse_str_val(str, pos + 1)
737:   elseif first == '-' or first:match('%d') then
738:     return parse_num_val(str, pos)
739:   elseif first == end_delim then
740:     return nil, pos + 1
741:   else  -- Parse true, false, or null.
742:     local literals = {['true'] = true, ['false'] = false, ['null'] = json.null}
743:     for lit_str, lit_val in pairs(literals) do
744:       local lit_end = pos + #lit_str - 1
745:       if str:sub(pos, lit_end) == lit_str then return lit_val, lit_end + 1 end
746:     end
747:     local pos_info_str = 'position ' .. pos .. ': ' .. str:sub(pos, pos + 10)
748:   end
